# Host tool confirmation popups (events/horsttool.txt hosttool.63 / hosttool.1000).
# Each click gives its request a token state (ht_issue_request_token, saved as the LOCAL event
# target ht_req_token that travels into the popup) and flags that state as the live request.
# Answering, Clear All and a host handoff all drop the flag, so a popup acts only on the exact
# request it was opened for. Evaluated in the popup's scope: the host country that clicked.

ht_annex_request_is_live = {
	has_country_flag = host
	has_country_flag = ht_annex_pending
	OR = {
		# the popup carries no token: fall back to the pending flag alone
		NOT = { has_event_target = ht_req_token }
		event_target:ht_req_token = { has_state_flag = ht_live_annex_req }
	}
}

ht_host_move_request_is_live = {
	has_country_flag = host
	has_country_flag = ht_host_move_pending
	OR = {
		NOT = { has_event_target = ht_req_token }
		event_target:ht_req_token = { has_state_flag = ht_live_move_req }
	}
}
