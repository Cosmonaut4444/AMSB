on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { has_game_rule = {rule = select_host_country option = GER} }
				set_variable = { host = GER.id }
			}
			else_if = {
				limit = { has_game_rule = {rule = select_host_country option = AST} }
				set_variable = { host = AST.id }
			}
			else_if = {
				limit = { has_game_rule = {rule = select_host_country option = BRA} }
				set_variable = { host = BRA.id }
			}
			else_if = {
				limit = { has_game_rule = {rule = select_host_country option = RAJ} }
				set_variable = { host = RAJ.id }
			}
			else_if = {
				limit = { has_game_rule = {rule = select_host_country option = BUL} }
				set_variable = { host = BUL.id }
			}
			else_if = {
				limit = { has_game_rule = {rule = select_host_country option = DEN} DEN = { exists = yes } }
				set_variable = { host = DEN.id }
			}
			else_if = {
				limit = { has_game_rule = {rule = select_host_country option = CAN} }
				set_variable = { host = CAN.id }
			}
			else_if = {
				limit = { has_game_rule = {rule = select_host_country option = FIN} }
				set_variable = { host = FIN.id }
			}
			else_if = {
				limit = { has_game_rule = {rule = select_host_country option = FRA} }
				set_variable = { host = FRA.id }
			}
			else_if = {
				limit = { has_game_rule = {rule = select_host_country option = ITA} }
				set_variable = { host = ITA.id }
			}
			else_if = {
				limit = { has_game_rule = {rule = select_host_country option = JAP} }
				set_variable = { host = JAP.id }
			}
			else_if = {
				limit = { has_game_rule = {rule = select_host_country option = HUN} }
				set_variable = { host = HUN.id }
			}
			else_if = {
				limit = { has_game_rule = {rule = select_host_country option = ROM} }
				set_variable = { host = ROM.id }
			}
			else_if = {
				limit = { has_game_rule = {rule = select_host_country option = MEX} }
				set_variable = { host = MEX.id }
			}
			else_if = {
				limit = { has_game_rule = {rule = select_host_country option = PHI} PHI = { exists = yes } }
				set_variable = { host = PHI.id }
			}
			else_if = {
				limit = { has_game_rule = {rule = select_host_country option = SAF} }
				set_variable = { host = SAF.id }
			}
			else_if = {
				limit = { has_game_rule = {rule = select_host_country option = SOV} }
				set_variable = { host = SOV.id }
			}
			else_if = {
				limit = { has_game_rule = {rule = select_host_country option = SPR} }
				set_variable = { host = SPR.id }
			}
			else_if = {
				limit = { has_game_rule = {rule = select_host_country option = ENG} }
				set_variable = { host = ENG.id }
			}
			else_if = {
				limit = { has_game_rule = {rule = select_host_country option = USA} }
				set_variable = { host = USA.id }
			}
			# Fallback: if the selected host country does not exist in this game mode
			# (SIA/PHI only exist at the normal-mode start), default the tool to GER
			# so the session always has a working host.
			if = {
				limit = { NOT = { has_variable = host } }
				set_variable={host = GER.id}
			}
			var:host = {
				set_variable = { host_marked_states = 0 }
				set_country_flag = host
			}
		}
	}
}
